function publicUser(u){ return u?{id:String(u.id),name:u.name,email:u.email,role:u.role,whatsapp:u.whatsapp||'',share_token:u.share_token||''}:null; }
async function getSession(req){
  const token=cookies(req)[SESSION_COOKIE]; if(!token) return null;
  const rows=await q(`SELECT u.id,u.name,u.email,u.role,u.active,u.whatsapp,u.share_token,s.token,s.expires_at
    FROM sessions s JOIN users u ON u.id=s.user_id
    WHERE s.token=$1 AND s.expires_at>$2 AND u.active=true LIMIT 1`,[token,nowSec()]);
  return rows[0]||null;
}
async function requireUser(req,res,roles=['ADMIN','ENCARGADO']){
  const u=await getSession(req); if(!u){err(res,401,'No autorizado');return null;}
  if(roles && !roles.includes(String(u.role).toUpperCase())){err(res,403,'Permiso insuficiente');return null;}
  return u;
}

function decodedSalt(s){
  if(/^[0-9a-f]{32,}$/i.test(s) && s.length%2===0) return Buffer.from(s,'hex');
  if(/^[A-Za-z0-9+/]+={0,2}$/.test(s) && s.length%4===0){try{return Buffer.from(s,'base64')}catch{}}
  return Buffer.from(s);
}
