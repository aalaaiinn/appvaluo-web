// AppValúo Croquis IA CAD · Convenio canónico 2D → 3D
// CAD: X derecha, Y arriba lógico. THREE: X derecha, Y altura, Z profundidad.
// Conversión oficial: X3D = Xcad, Y3D = elevación, Z3D = -Ycad.
function cadToWorld(pt={}, elevation=0){
  return {x:Number(pt.x)||0,y:Number(elevation)||0,z:-(Number(pt.y)||0)};
}
function worldToCad(v={}){
  return {x:Number(v.x)||0,y:-(Number(v.z)||0)};
}
function cadAngleToWorld(angleRad=0){
  // Three.js rotation.y positiva lleva +X hacia -Z, exactamente el giro positivo del CAD (+Y lógico).
  return Number(angleRad)||0;
}
function cadDegreesToWorld(deg=0){
  return (Number(deg)||0)*Math.PI/180;
}
function worldAngleToCad(angleRad=0){
  return Number(angleRad)||0;
}
function cadPolygonToXZ(poly=[]){
  return (Array.isArray(poly)?poly:[]).map(p=>({x:Number(p.x)||0,z:-(Number(p.y)||0)}));
}
function cadNorthAngleToWorld(angleDegClockwiseFromCadUp=0){
  const r=(Number(angleDegClockwiseFromCadUp)||0)*Math.PI/180;
  // 0° = +Y CAD => -Z mundo. 90° = +X CAD => +X mundo.
  return {x:Math.sin(r),y:0,z:-Math.cos(r)};
}
function transformedBounds(points=[]){
  const w=(Array.isArray(points)?points:[]).map(p=>cadToWorld(p,0));
  if(!w.length)return{minX:-3,maxX:3,minZ:-5,maxZ:5,cx:0,cz:0,w:6,d:10};
  const xs=w.map(p=>p.x),zs=w.map(p=>p.z),minX=Math.min(...xs),maxX=Math.max(...xs),minZ=Math.min(...zs),maxZ=Math.max(...zs);
  return{minX,maxX,minZ,maxZ,cx:(minX+maxX)/2,cz:(minZ+maxZ)/2,w:Math.max(.1,maxX-minX),d:Math.max(.1,maxZ-minZ)};
}
